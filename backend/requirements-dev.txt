# Development / CI only — deliberately NOT installed in production.
#
# render.yaml's buildCommand installs requirements.txt alone, so nothing here
# reaches a deployed image. Keeping these out of the production dependency
# tree removes their transitive attack surface entirely:
#   fastmcp -> fastmcp-slim[server] -> joserfc  (3 CVEs) + authlib
#   pytest  -> PYSEC-2026-1845
# None of these packages are imported by app/ — mcp/fastmcp are used only by
# the local admin tooling (roomivo_mcp.py, test_mcp_client.py) and aiohttp
# only by tests/comprehensive_test.py.

-r requirements.txt

# Test runner. pytest >=9.0.3 closes PYSEC-2026-1845.
# pytest-asyncio 1.x changed the default event-loop scope, surfacing a
# pre-existing cross-test leak: tests/test_intl_rails.py (10 tests) passes in
# isolation but fails under the full `pytest tests/` run with "no current
# event loop in thread 'MainThread'". Test-suite migration, not a dependency
# problem — tracked as its own follow-up, not blocking the CVE fix.
pytest>=9.0.3
pytest-asyncio>=1.0
aiohttp>=3.10.0

# Local MCP admin tooling (roomivo_mcp.py) — never run in prod or CI
mcp>=1.0.0
fastmcp>=0.1.0
